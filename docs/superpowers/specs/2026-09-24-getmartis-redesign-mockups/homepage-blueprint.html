Declare the resource. Ship the workflow.

Martis turns plain PHP classes into a React workspace with routes, validation, relationships and persistent view state.

ClientResource.phpPHP
class ClientResource extends Resource
{
  public function fields(): array
  {
    return [
      Text::make('name')->searchable(),
      Select::make('plan')->options(...),
      Currency::make('revenue'),
      Badge::make('status'),
    ];
  }
}
